### Updated JavaScript (script.js)

```javascript
const player = document.getElementById('player');
const obstacle = document.getElementById('obstacle');
const scoreText = document.getElementById('score');
const restartBtn = document.getElementById('restartBtn');

let isJumping = false;
let isGameOver = false;
let score = 0;

function jump() {
    if (isJumping) return;
    isJumping = true;

    let jumpHeight = 0;
    const jumpInterval = setInterval(() => {
        if (jumpHeight >= 100) {
            clearInterval(jumpInterval);
            fall();
        } else {
            jumpHeight += 5;
            player.style.bottom = `${20 + jumpHeight}px`;
        }
    }, 20);
    score++; // Increment score for each jump
    scoreText.innerText = `Score: ${score}`; // Update score display
}

function fall() {
    let fallHeight = 100;
    const fallInterval = setInterval(() => {
        if (fallHeight <= 0) {
            clearInterval(fallInterval);
            isJumping = false;
        } else {
            fallHeight -= 5;
            player.style.bottom = `${20 + fallHeight}px`;
        }
    }, 20);
}

function checkCollision() {
    const playerRect = player.getBoundingClientRect();
    const obstacleRect = obstacle.getBoundingClientRect();

    if (
        playerRect.x < obstacleRect.x + obstacleRect.width &&
        playerRect.x + playerRect.width > obstacleRect.x &&
        playerRect.y < obstacleRect.y + obstacleRect.height &&
        playerRect.y + playerRect.height > obstacleRect.y
    ) {
        alert('Game Over!');
        clearInterval(obstacleMove);
        isGameOver = true;
        restartBtn.classList.remove('hidden'); // Show the restart button
    }
}

let obstacleMove = setInterval(() => {
    if (isGameOver) return;

    let obstaclePosition = parseInt(window.getComputedStyle(obstacle).getPropertyValue('right'));
    obstaclePosition += 2;

    if (obstaclePosition > 600) {
        obstaclePosition = 0; // Reset position
    }
    
    obstacle.style.right = `${obstaclePosition}px`;
    checkCollision();
}, 20);

document.addEventListener('keydown', (event) => {
    if (event.code === 'Space' && !isGameOver) {
        jump();
    }
});

// Restart the game
restartBtn.addEventListener('click', () => {
    isGameOver = false;
    score = 0;
    scoreText.innerText = `Score: 0`; // Reset score display
    obstacle.style.right = '0'; // Reset obstacle position
    player.style.bottom = '20px'; // Reset player position
    restartBtn.classList.add('hidden'); // Hide the restart button

    // Restart the obstacle movement
    obstacleMove = setInterval(() => {
        let obstaclePosition = parseInt(window.getComputedStyle(obstacle).getPropertyValue('right'));
        obstaclePosition += 2;

        if (obstaclePosition > 600) {
            obstaclePosition = 0;
        }

        obstacle.style.right = `${obstaclePosition}px`;
        checkCollision();
    }, 20);
});
```

### Explanation of New Features:

1. **Score Tracking:**
   - A score variable (`score`) increments each time the player successfully jumps.
   - The score is displayed in the top left corner using the `score` div, which updates every time the player jumps.

2. **Restart Button:**
   - A button with the text "Play Again" is added. This button is initially hidden and shows up when the game is over.
   - Clicking the button resets the game state, including the score, player position, and obstacle position.

3. **Game Over Handling:**
   - Once a collision is detected, the game stops, and the restart button becomes visible.
   - The user can click the "Play Again" button to restart the game without refreshing the page.
